cmake_minimum_required(VERSION 3.17)

project(lua C)

file(GLOB_RECURSE HDR_LIBLUA *.h)

add_library(lualib STATIC onelua.c)
target_compile_definitions(lualib PRIVATE MAKE_LIB)
set_target_properties(lualib PROPERTIES PUBLIC_HEADER "${HDR_LIBLUA}")

include(GNUInstallDirs)

install(TARGETS lualib
    PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/Lua
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/release
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}/release
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}/release)
