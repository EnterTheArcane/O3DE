# Model the engine's staging property in the standalone fixture.
function(ly_add_target_files)
    cmake_parse_arguments(PARSE_ARGV 0 arg "" "OUTPUT_SUBDIRECTORY" "TARGETS;FILES")
    foreach(target IN LISTS arg_TARGETS)
        foreach(file IN LISTS arg_FILES)
            set_property(TARGET "${target}" APPEND PROPERTY INTERFACE_LY_TARGET_FILES
                "${file}\n${arg_OUTPUT_SUBDIRECTORY}")
        endforeach()
    endforeach()
endfunction()

function(ly_target_include_system_directories)
    cmake_parse_arguments(PARSE_ARGV 0 arg "" TARGET INTERFACE)
    target_include_directories("${arg_TARGET}" SYSTEM INTERFACE ${arg_INTERFACE})
endfunction()
