cmake_minimum_required(VERSION 3.25)
project(PhysXTestPackage LANGUAGES CXX)

find_package(PhysX 5.13.0 EXACT CONFIG REQUIRED)
get_target_property(physx_configs PhysX::PhysX IMPORTED_CONFIGURATIONS)
if(PHYSX_TEST_MULTI_CONFIG)
    list(SORT physx_configs)
    if(NOT physx_configs STREQUAL "CHECKED;DEBUG;PROFILE;RELEASE")
        message(FATAL_ERROR "Expected four PhysX configurations, found ${physx_configs}")
    endif()
    foreach(config IN LISTS physx_configs)
        string(TOLOWER "${config}" directory)
        get_target_property(physx_library PhysX::PhysX "IMPORTED_LOCATION_${config}")
        if(NOT physx_library MATCHES "/lib/${directory}/[^/]+$")
            message(FATAL_ERROR "Expected PhysX library in lib/${directory}: ${physx_library}")
        endif()
    endforeach()
else()
    list(LENGTH physx_configs physx_config_count)
    if(NOT physx_config_count EQUAL 1)
        message(FATAL_ERROR "Expected one PhysX configuration, found ${physx_configs}")
    endif()
    get_target_property(physx_library PhysX::PhysX "IMPORTED_LOCATION_${physx_configs}")
    if(NOT physx_library MATCHES "/lib/[^/]+$")
        message(FATAL_ERROR "Expected PhysX library directly in lib: ${physx_library}")
    endif()
endif()
if(PHYSX_TEST_CUDA AND NOT TARGET PhysX::PhysXGpu)
    message(FATAL_ERROR "The CUDA-enabled package must export PhysX::PhysXGpu")
endif()
add_executable(test_package test.cpp)
target_compile_features(test_package PRIVATE cxx_std_20)
target_compile_definitions(test_package PRIVATE "$<IF:$<CONFIG:Debug>,_DEBUG,NDEBUG>")
target_link_libraries(test_package PRIVATE PhysX::physx_lib)
